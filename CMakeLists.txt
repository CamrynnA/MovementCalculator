cmake_minimum_required(VERSION 3.21)
project(MovementCalculator LANGUAGES CXX RC)
add_executable(MovementCalculator WIN32 main.cpp movement_calculator.cpp resource.rc)
target_compile_features(MovementCalculator PRIVATE cxx_std_20)
target_compile_definitions(MovementCalculator PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(MovementCalculator PRIVATE dwmapi comctl32)
if(MSVC)
  target_compile_options(MovementCalculator PRIVATE /W4 /permissive- /utf-8)
endif()
